## Application

```go [main.go]
app := tako.NewApp()
app.Mount(adapter)
tako.Run(app)
```

## Service Container

```go [main.go]
// Register
app.Container().Singleton(new(contracts.Logger), myLogger)
app.Container().Lazy(new(*APIClient), func() (any, error) { return NewClient(), nil })

// Resolve
var logger contracts.Logger
app.Container().Make(&logger)

// Resolve with generics
bus, _ := foundation.Make[contracts.EventBus](app)
// or panic on error
bus := foundation.MustMake[contracts.EventBus](app)
```

## Keys

```go [example.go]
// Global
ctx.Keys().Bind("ctrl+q", func() { app.Shutdown() }).Desc("Quit application")
ctx.Keys().Bind("?|ctrl+/", showHelp).Desc("Show help")

// Zone-scoped
zone := ctx.Keys().Zone("sidebar")
zone.Bind("j|down", moveDown).Desc("Move down")
zone.Bind("k|up", moveUp).Desc("Move up")
```

## Mouse

```go [example.go]
// Declare hitbox every render (in your layout)
ctx.Mouse().UpdateHitbox("sidebar", x, y, width, height)

// Handle events (in your provider)
ctx.Mouse().Zone("sidebar").OnClick(func(x, y int) { selectItem(y) })
ctx.Mouse().Zone("sidebar").OnScrollDown(func(x, y int) { scrollList(1) })
```

## Events

```go [example.go]
// Emit
ctx.Emit("order:created", order)

// Subscribe (auto-cleans on provider shutdown)
ctx.Subscribe(ctx.Context(), "order:created", func(e contracts.Event) {
    order := e.Data.(*Order)
})

// Unsubscribe manually
unsub := ctx.Events().On("tick", handler)
unsub()
```

## State

```go [example.go]
// Simple set/get
ctx.State().Set("cart.count", 5)
count := ctx.State().Get("cart.count")

// Reactive observer
watcher := ctx.State().Watch("cart.total")
watcher.Debounce(100 * time.Millisecond)
watcher.OnUpdate(func(old, new any) { refreshBadge(new) })
watcher.Subscribe(ctx.Context())
```

## RPC (Request/Response)

```go [example.go]
// Register handler
route := ctx.RPC().Route("users.fetch")
route.Timeout(5 * time.Second)
route.Handle(func(c context.Context, req contracts.RPCRequest) (contracts.RPCResponse, error) {
    user := fetchUser(req.Payload.(string))
    return contracts.RPCResponse{Data: user}, nil
})

// Call
call := ctx.RPC().Call("users.fetch")
call.WithPayload("user-123")
resp, err := call.Await()
```

## Providers

```go [providers/provider.go]
type Provider struct{}

func (p *Provider) Boot(app *foundation.Application) error {
    ctx := app.Context()
    ctx.Keys().Bind("ctrl+f", p.openSearch)
    ctx.Subscribe(ctx.Context(), "order:created", p.notify)
    return nil
}
```

## Overlays

```go [example.go]
// Basic
ctx.UI().Show("search", func() any { return searchView })
ctx.UI().Unmount()

// Dialog
ctx.Dialog().Confirm("Delete this order?", func(yes bool) {
    if yes { deleteOrder() }
})
```

## Hooks

```go [example.go]
// Provide (from provider)
ctx.Hooks().Set("render:toolbar", func() any { return toolbarView })
ctx.Hooks().Add("app.sidebar", func() any { return sidebarItem })

// Consume (from layout)
toolbar := hooks.Get("render:toolbar")
items := hooks.All("app.sidebar")
```

## Scheduler

```go [example.go]
// Background job
job := ctx.Jobs().Dispatch(func(c context.Context) (any, error) {
    return fetchAPI(c)
})
job.OnSuccess(func(data any) {
    ctx.State().Set("api.data", data)
})

// Periodic
stop := ctx.Jobs().Every(30*time.Second, func() { syncData() })
```

## Focus Stack

```go [example.go]

```


## Storage (Persistent KV)

```go [example.go]
ctx.Storage().Set("preferences.theme", "dracula")

var theme string
ctx.Storage().Get("preferences.theme", &theme)

ctx.Storage().Delete("preferences.theme")
```

## Theming

```go [example.go]
ctx.Theme().Register("dracula", map[string]string{
    "primary": "#bd93f9", "bg": "#282a36",
})
ctx.Theme().Use("dracula")
color := ctx.Theme().Get("primary")
```

## i18n

```go [example.go]
ctx.Lang().Register("id_ID", map[string]string{"greeting": "Halo, %s!"})
ctx.Lang().SetLocale("id_ID")
msg := ctx.Lang().T("greeting", "Budi") // "Halo, Budi!"
```
